# -----------------------------
# Inputs / initialization
# -----------------------------
Full dataset = D

Split once:
    Training set D_tr = 80%
    Validation set D_val = 20% (half of it will be the holdout set)
    # stratified so each difficulty level appears in validation

Define buckets on training set:
    {I_k}_{k=1}^K
    # e.g. easy, medium, hard based on difficulty labels

Initialize:
    student parameters θ_0
    teacher parameters φ

Set hyperparameters (PoC): 
    episodes    = 100                           
    T           = 20        # start with 20-50 so the credit assignment is not hard with long horizons
    B           = 32        # or 64
    m           = 32          
    η_student   = 10^(-4) # 0.99 is too high                           
    γ           = ...
    λ_cost      = 0 
    α_reg       = ...
    p_ref       = ...
    hidden_dim  = 64 - then if teacher not learning well (128), if training too slow or overfitting (go to 32)
    

Initialize running bucket statistics:
    avg_loss_per_bucket
    change_in_loss_per_bucket


for each episode: 

    initialize student
    for t in 0 ... T-1:

        # 1. Construct teacher observation
        o_t = (
            current step t,
            running average loss per bucket,
            running change in loss per bucket
        )

        # 2. Teacher chooses action (distribution over buckets) -> teacher policy μ_φ(. | o_t) 
        logits_t = TeacherMLP_φ(o_t)
        teacher_probs_t = softmax(logits_t)                             # teacher_probs_t is the mean/preferred bucket distribution
        concentration_t = concentration_scale * teacher_probs_t + ε     # concentration_scale controls how noisy/exploratory the sampled action is
        a_t ~ Dirichlet(concentration_t)                                # a_t is the actual sampled distribution over buckets used at this step
        log_prob_t = log Dirichlet(a_t ; concentration_t)               # stored for the REINFORCE update


        # 3-4. Sample training mini-batch B_t from D_tr using a_t
        training_batch = []
        for b in 1 ... B: # for each example in the batch
            sample bucket k_{t,b} ~ Categorical(a_t)            # Pick which bucket this example should come from.
            sample example index i_{t,b} ~ Uniform(I_{k_{t,b}}) # Then pick a random example inside that bucket.
            add example e_{i_{t,b}} to training_batch
            # so batch B matches the distribution dictated by a_t

        # 5. Sample one validation mini-batch from fixed D_val
        sample validation batch B̃_t of size m uniformly from D_val

        # 6. Compute validation loss BEFORE learner update
        val_loss_before = average SFT loss on B̃_t using θ_t

        # 7. Train learner on current training batch
        for each example in training_batch:
            student predicts answer
            compute SFT loss (cross-entropy against target)
            compute gradient w.r.t. θ_t
        average gradients across the B examples

        # 8. Learner update (SFT)
        θ_{t+1} = θ_t - η_student * averaged_gradient

        # 9. Update running bucket statistics (only for buckets that appeared in the current training batch) 
        compute per-bucket batch loss
        update running average loss per bucket
        update running change in loss per bucket

        # 10. Compute validation loss AFTER learner update
        val_loss_after = average SFT loss on SAME validation batch B̃_t using θ_{t+1}

        # 11. Compute reward
        validation_improvement = val_loss_before - val_loss_after
        compute_cost = c(training_batch)
        stability_penalty = KL(a_t || p_ref)

        r_t = validation_improvement - λ_cost * compute_cost - α_reg * stability_penalty

        store r_t in trajectory
        store log_prob_t = log μ_φ(a_t | o_t) in trajectory (REINFORCE is episodic)

    # End training -> episode ended (because you accumulate then update)
    
    for t in range(T): # Compute all returns 
        q_t = sum_{k=t}^{T-1} γ^{k-t} r_k       # r_k = reward at time step k
    
    episode_baseline = mean (q_0, q_1, ..., q_{T-1})    # over the steps in an episode
    for t in range(T): # compute advantages A_t
        A_t = q_t - episode_baseline
    
    # Compute policy gradient
    total_gradient = 0
    for t in range(T):
        total_gradient += ∇_φ log μ_φ(a_t | o_t) * A_t 

    # 12. Teacher update once per episode using all stored log_probs and rewards
    φ ← φ + η_teacher * total_gradient
